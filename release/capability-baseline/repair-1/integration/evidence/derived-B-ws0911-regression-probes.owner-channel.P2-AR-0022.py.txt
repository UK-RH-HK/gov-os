def repo(tag, files):
    os.makedirs(synth.SCRATCH, exist_ok=True)
    base = os.path.join(synth.SCRATCH, f"{tag}-{uuid.uuid4().hex[:6]}")
    root, home = os.path.join(base, "proj"), os.path.join(base, "home")
    os.makedirs(root)
    os.makedirs(home)
    for rel, text in files.items():
        p = os.path.join(root, rel)
        os.makedirs(os.path.dirname(p), exist_ok=True)
        open(p, "w").write(text)
    git(root, "init", "-q")
    git(root, "config", "user.email", "b@example.invalid")
    git(root, "config", "user.name", "b")
    commit(root, "baseline")
    log(f"# repo {tag}: {root}")
    return root, Gov(root, home)


def cat(root):

def rd(root, rel):

# a human answer is an owner-signed document relayed by a declared L3 role, never `--role human --by owner`; the owner
# is WS-3's test-material reference signer (repair-1/ws03/evidence/hc_owner.py, published seed), anchored on this
# unprovisioned probe machine by a standalone anchor placed outside the repository.
HC_OWNER = os.path.join(WT, "release", "capability-baseline", "repair-1", "ws03", "evidence", "hc_owner.py")


def answer_all(g, root, option):
    ex = g.as_(role="migration-executor", session="S-exec")
    st = ex.run("trust", "human-channel", quiet=True)
    if not (st.get("ok") and (st.get("result") or {}).get("available")):
        af = os.path.join(os.path.dirname(root), f"anchor-{uuid.uuid4().hex[:6]}.json")
        subprocess.run([sys.executable, HC_OWNER, "anchor", af, "7"], check=True, capture_output=True)
        ex.run("trust", "human-channel", "--provision", af, quiet=True)
    for e in cat(root):
        if e.get("requires_human_gate") and e.get("human_gate"):
            r = ex.run("gate", "present", e["human_gate"], quiet=True)
            gg = (r.get("result") or {}).get("gate") or {}
            f = os.path.join(os.path.dirname(root), f"answer-{e['human_gate']}-{uuid.uuid4().hex[:6]}.json")
            subprocess.run([sys.executable, HC_OWNER, "answer", f, e["human_gate"], gg.get("gate_instance", ""),
                            gg.get("package_sha256", ""), option], check=True, capture_output=True)
            ex.run("decide", e["human_gate"], "--option", option, "--answer-file", f, quiet=True)
